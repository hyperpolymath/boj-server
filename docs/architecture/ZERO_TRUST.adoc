= Zero Trust Architecture for BOJ Server
:toc: preamble
:icons: font

[TIP]
====
**Zero Trust Status:** This document outlines the Zero Trust Architecture roadmap for BOJ Server Mistral. Current implementation uses perimeter-based security with Zero Trust design patterns in development.
====

== Why Zero Trust Matters for BOJ Server

**Threat Model:**
* Traditional perimeter security assumes internal networks are trusted
* Compromised nodes can move laterally within the network
* Insider threats have excessive access by default
* Cloud and hybrid environments blur network boundaries
* BOJ Server handles sensitive code submissions requiring strict access control

**Current Security Gaps:**
* No mutual TLS (mTLS) for internal node communication
* Implicit trust between BoJ nodes
* No continuous authentication for Stapeln ↔ BoJ interactions
* Gossip network lacks node identity verification
* Over-reliance on network-level security

    }

// src/network/mtls/client.rs
use rustls::{ClientConfig, RootCertStore};
use std::sync::Arc;

pub struct MtlsClient {
    client_config: Arc<ClientConfig>,
    identity: Arc<dyn Identity>,
}

impl MtlsClient {
    pub fn new(identity: Arc<dyn Identity>, ca_certs: Vec<Certificate>) -> Result<Self> {
        let mut root_store = RootCertStore::empty();
        for cert in ca_certs {
            root_store.add(&cert)?;
        }
        
        let mut config = ClientConfig::builder()
            .with_safe_defaults()
            .with_root_certificates(root_store)
            .with_client_auth_cert_resolver(Arc::new(IdentityResolver::new(identity.clone())))?;
        
        // Require server certificate validation
        config.dangerous().set_certificate_verifier(Arc::new(
            WebPkiVerifier::new(root_store, None)
        ));
        
        Ok(Self {
            client_config: Arc::new(config),
            identity,
        })
    }
    
    pub async fn connect(&self, domain: &str) -> Result<ClientConnection> {
        let server_name = ServerName::try_from(domain)?
            .to_owned();
        
        let conn = ClientConnection::new(Arc::clone(&self.client_config), server_name)?;
        Ok(conn)
    }
}

// src/network/mtls/server.rs
use rustls::{ServerConfig, NoClientAuth, RootCertStore};

pub struct MtlsServer {
    server_config: Arc<ServerConfig>,
    identity: Arc<dyn Identity>,
}

impl MtlsServer {
    pub fn new(
        identity: Arc<dyn Identity>,
        ca_certs: Vec<Certificate>,
        client_auth: ClientAuth
    ) -> Result<Self> {
        let mut root_store = RootCertStore::empty();
        for cert in ca_certs {
            root_store.add(&cert)?;
        }
        
        let client_auth = match client_auth {
            ClientAuth::Required => {
                let verifier = WebPkiVerifier::new(root_store, None);
                AllowAnyAnonymousOrAuthenticatedClient::new(Arc::new(verifier))
            }
            ClientAuth::Optional => NoClientAuth::new(),
        };
        
        let mut config = ServerConfig::builder()
            .with_safe_defaults()
            .with_client_cert_verifier(client_auth)
            .with_single_cert(identity.cert_chain(), identity.key())?;
        
        Ok(Self {
            server_config: Arc::new(config),
            identity,
        })
    }
    
    pub fn accept(&self, stream: TcpStream) -> Result<ServerConnection> {
        let conn = ServerConnection::new(Arc::clone(&self.server_config))?;
        Ok(conn)
    }
}

// src/network/gossip/secure.rs
pub struct SecureGossipNode {
    address: String,
    mtls_client: MtlsClient,
    node_identity: Arc<dyn Identity>,
    policy_engine: Arc<dyn PolicyEngine>,
}

impl SecureGossipNode {
    pub fn new(
        address: String,
        mtls_client: MtlsClient,
        identity: Arc<dyn Identity>,
        policy_engine: Arc<dyn PolicyEngine>
    ) -> Self {
        Self {
            address,
            mtls_client,
            node_identity: identity,
            policy_engine,
        }
    }
    
    pub async fn send_message(&self, message: &GossipMessage) -> Result<()> {
        // Step 1: Establish mTLS connection
        let domain = extract_domain(&self.address);
        let mut conn = self.mtls_client.connect(domain).await?;
        
        // Step 2: Perform policy check
        let peer_identity = conn.peer_identity()?;
        self.policy_engine.can_send_to(
            &self.node_identity,
            &peer_identity,
            message.message_type()
        )?;
        
        // Step 3: Sign the message
        let signed_message = self.sign_message(message)?;
        
        // Step 4: Send encrypted message
        conn.write_all(&signed_message.serialize()?)?;
        
        Ok(())
    }
    
    fn sign_message(&self, message: &GossipMessage) -> Result<SignedGossipMessage> {
        let signature = self.node_identity.sign(&message.serialize()?)?;
        Ok(SignedGossipMessage {
            message: message.clone(),
            signature,
            timestamp: Utc::now(),
            node_id: self.node_identity.id().to_string(),
        })
    }
    
    pub async fn receive_message(&self, stream: TcpStream) -> Result<GossipMessage> {
        // Step 1: Establish mTLS connection
        let mut conn = self.mtls_client.accept(stream)?;
        
        // Step 2: Get peer identity
        let peer_identity = conn.peer_identity()?;
        
        // Step 3: Read and verify message
        let signed_message = conn.read_signed_message()?;
        
        // Step 4: Verify signature
        peer_identity.verify(&signed_message.message, &signed_message.signature)?;
        
        // Step 5: Policy check
        self.policy_engine.can_receive_from(
            &peer_identity,
            &self.node_identity,
            signed_message.message.message_type()
        )?;
        
        Ok(signed_message.message)
    }
}

// src/elixir/stapeln/secure.rs
pub struct SecureStapelnConnection {
    mtls_client: MtlsClient,
    identity: Arc<dyn Identity>,
    policy_engine: Arc<dyn PolicyEngine>,
    service_discovery: Arc<dyn ServiceDiscovery>,
}

impl SecureStapelnConnection {
    pub async fn multiply(&self, problem: &Problem) -> Result<MultipliedProblem> {
        // Step 1: Discover Stapeln service with current security posture
        let stapeln_service = self.service_discovery.discover("stapeln")
            .await?;
        
        // Step 2: Check policy - can we connect to this Stapeln instance?
        self.policy_engine.can_connect_to(
            &self.identity,
            &stapeln_service.identity,
            "multiply"
        )?;
        
        // Step 3: Establish mTLS connection
        let mut conn = self.mtls_client.connect(&stapeln_service.address)
            .await?;
        
        // Step 4: Create signed request
        let request = self.create_signed_request(problem)?;
        
        // Step 5: Send request and verify response
        let response = conn.send_request(&request).await?;
        self.verify_response(&response)?;
        
        Ok(response.problem)
    }
    
    fn create_signed_request(&self, problem: &Problem) -> Result<SignedRequest> {
        let request_id = Uuid::new_v4();
        let request = Request {
            id: request_id,
            problem: problem.clone(),
            timestamp: Utc::now(),
            caller: self.identity.id().to_string(),
        };
        
        let signature = self.identity.sign(&request.serialize()?)?;
        
        Ok(SignedRequest {
            request,
            signature,
            request_certificate: self.identity.cert().clone(),
        })
    }
    
    fn verify_response(&self, response: &SignedResponse) -> Result<()> {
        // Verify response signature
        response.identity.verify(
            &response.response,
            &response.signature
        )?;
        
        // Check response freshness
        if response.response.timestamp < Utc::now() - ChronoDuration::minutes(5) {
            return Err(SecurityError::StaleResponse);
        }
        
        // Verify response matches request
        if response.response.request_id != self.current_request_id {
            return Err(SecurityError::InvalidResponse);
        }
        
        Ok(())
    }
}

        &self,
        source: &dyn Identity,
        destination: &dyn Identity,
        message_type: &str
    ) -> Result<bool> {
        let input = serde_json::json!({
            "action": "gossip",
            "source": {
                "identity": source.claims(),
                "security_level": source.security_level(),
            },
            "destination": {
                "identity": destination.claims(),
                "security_level": destination.security_level(),
            },
            "message": {
                "type": message_type,
            }
        });
        
        let result = self.client.evaluate(
            "boj/access/allow",
            &input
        )?;
        
        if !result.allow {
            return Err(PolicyError::AccessDenied {
                source: source.id().to_string(),
                destination: destination.id().to_string(),
                action: "gossip".to_string(),
            });
        }
        
        Ok(true)
    }

// src/security/policy/continuous.rs
pub struct ContinuousAuthorizer {
    policy_engine: Arc<dyn PolicyEngine>,
    risk_engine: Arc<dyn RiskEngine>,
    session_store: Arc<dyn SessionStore>,
}

impl ContinuousAuthorizer {
    pub async fn authorize_request(&self, request: &Request) -> Result<AuthorizationDecision> {
        // Step 1: Initial policy check
        let initial_decision = self.evaluate_policy(request).await?;
        if !initial_decision.allowed {
            return Ok(initial_decision);
        }
        
        // Step 2: Risk assessment
        let risk_score = self.evaluate_risk(request).await?;
        
        // Step 3: Continuous monitoring setup
        let session_id = self.session_store.create_session(
            request,
            risk_score,
            initial_decision.policies
        ).await?;
        
        Ok(AuthorizationDecision {
            allowed: true,
            session_id,
            risk_score,
            policies: initial_decision.policies,
            expires_at: Utc::now() + ChronoDuration::minutes(5),
        })
    }
    
    pub async fn reauthorize_session(&self, session_id: &str) -> Result<ReauthorizationDecision> {
        let session = self.session_store.get_session(session_id).await?;
        
        // Check if session is still valid
        if session.expires_at < Utc::now() {
            return Ok(ReauthorizationDecision::Denied("Session expired".to_string()));
        }
        
        // Re-evaluate risk
        let current_risk = self.evaluate_risk(&session.original_request).await?;
        
        if current_risk > session.risk_score + 20.0 { // 20% increase threshold
            return Ok(ReauthorizationDecision::Denied(
                format!("Risk score increased from {} to {}", session.risk_score, current_risk)
            ));
        }
        
        // Extend session
        self.session_store.extend_session(session_id).await?;
        
        Ok(ReauthorizationDecision::Allowed {
            expires_at: Utc::now() + ChronoDuration::minutes(5),
            new_risk_score: current_risk,
        })
    }
    
    async fn evaluate_risk(&self, request: &Request) -> Result<f32> {
        let behavioral_score = self.risk_engine.evaluate_behavior(request).await?;
        let threat_intel_score = self.risk_engine.evaluate_threat_intel(request).await?;
        let anomaly_score = self.risk_engine.evaluate_anomalies(request).await?;
        
        Ok(behavioral_score * 0.5 + threat_intel_score * 0.3 + anomaly_score * 0.2)
    }
}

        vec![server_identity.cert().clone()]
    ).unwrap();

    let server_task = tokio::spawn(async move {
        let mut conn = server.accept(server_stream).unwrap();
        conn.read_message().await.unwrap()
    });
    
    let client_task = tokio::spawn(async move {
        let mut conn = client.connect("test.example.com").await.unwrap();
        conn.write_message(b"test message").await.unwrap()
    });

   let mtls_client = MtlsClient::new(identity.clone(), ca_certs)?;

   ```rust
   if !policy_engine.can_perform_action(&identity, "read-problems", &resource) {
       return Err(AuthorizationError::Denied);
   }
   ```

   ```rust
   let auth_result = continuous_authorizer.authorize_request(&request).await?;
   if !auth_result.allowed {
       return Err(AuthorizationError::SessionExpired);
   }
   ```

== Appendix: Policy Examples

**Gossip Protocol Policy:**
[source,rego]
--
# Allow gossip only between nodes with valid Svalinn attestation
allow {
    input.action == "gossip"
    input.source.identity.svalinn_attested == true
    input.destination.identity.svalinn_attested == true
    input.source.identity.security_level >= 2
    input.destination.identity.security_level >= 2
    same_trust_domain(input.source.identity, input.destination.identity)
}

same_trust_domain(a, b) {
    a.trust_domain == b.trust_domain
}

**Key Rotation Policy:**
[source,rego]
--
# Only security team can rotate keys
allow {
    input.action == "rotate-key"
    input.source.identity.groups[_] == "security-team"
    input.source.identity.mfa_completed == true
    input.destination.identity.service == "key-manager"
    # Require approval from another security team member
    count(input.approvals) > 0
    all_approvers_valid(input.approvals)
}

all_approvers_valid(approvers) {
    approvers[i].identity.groups[_] == "security-team"
    approvers[i].identity.mfa_completed == true
    approvers[i].timestamp >= time_now - 300  # Approval within last 5 minutes
}
