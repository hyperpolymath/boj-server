== Why a Cartridge Marketplace Matters

**Problem Statement:**
* Users struggle to discover useful cartridges
* No quality signals for cartridge selection
* Difficult to find cartridges for specific use cases
* Lack of community feedback and ratings
* No curated recommendations for beginners

**Marketplace Goals:**
* **Discovery:** Help users find the right cartridge quickly
* **Quality:** Surface high-quality, well-maintained cartridges
* **Trust:** Provide verification and security information
* **Community:** Enable user feedback and recommendations
* **Monetization:** Support cartridge developers

// src/services/search.rs
pub struct SearchService {
    elastic_client: Elasticsearch,
    metadata_db: PgPool,
}

impl SearchService {
    pub async fn search(
        &self,
        query: &str,
        category: Option<&str>,
        limit: usize,
        offset: usize
    ) -> Result<SearchResult> {
        // Build Elasticsearch query
        let mut es_query = ElasticsearchQuery::new();
        
        if !query.is_empty() {
            es_query = es_query.with_query(
                Query::multi_match([
                    ("name", query),
                    ("description", query),
                    ("content", query),
                ])
            );
        }
        
        if let Some(cat) = category {
            es_query = es_query.with_filter(
                Filter::term("categories", cat)
            );
        }
        
        // Execute search
        let es_results = self.elastic_client.search(es_query)
            .from(offset)
            .size(limit)
            .execute().await?;
        
        // Enrich with metadata
        let cartridge_ids: Vec<_> = es_results.hits.iter()
            .map(|hit| hit.id.clone())
            .collect();
        
        let metadata = self.get_metadata_batch(&cartridge_ids).await?;
        
        Ok(SearchResult {
            results: es_results.hits.into_iter()
                .map(|hit| self.enrich_result(hit, &metadata))
                .collect(),
            total: es_results.total,
            limit,
            offset,
        })
    }
    
    async fn get_metadata_batch(&self, ids: &[String]) -> Result<HashMap<String, CartridgeMetadata>> {
        // Query PostgreSQL for additional metadata
        let rows = sqlx::query!(
            "SELECT id, stars, forks, downloads, rating, is_verified FROM cartridges WHERE id = ANY($1)",
            &ids
        )
        .fetch_all(&self.metadata_db)
        .await?;
        
        Ok(rows.into_iter()
            .map(|row| (
                row.id,
                CartridgeMetadata {
                    stars: row.stars,
                    forks: row.forks,
                    downloads: row.downloads,
                    rating: row.rating,
                    is_verified: row.is_verified,
                }
            ))
            .collect())
    }
}

pub struct EngagementService {
    db: PgPool,
}

impl EngagementService {
    pub async fn star_cartridge(&self, user_id: &str, cartridge_id: &str) -> Result<bool> {
        let result = sqlx::query!(
            "INSERT INTO cartridge_stars (user_id, cartridge_id) VALUES ($1, $2)
             ON CONFLICT (user_id, cartridge_id) DO NOTHING",
            user_id,
            cartridge_id
        )
        .execute(&self.db)
        .await?;
        
        // Update cartridge star count
        self.update_star_count(cartridge_id).await?;
        
        Ok(result.rows_affected() > 0)
    }
    
    pub async fn fork_cartridge(&self, user_id: &str, cartridge_id: &str) -> Result<bool> {
        // Create a fork record
        let fork_id = Uuid::new_v4().to_string();
        
        sqlx::query!(
            "INSERT INTO cartridge_forks (id, user_id, cartridge_id, created_at)
             VALUES ($1, $2, $3, NOW())",
            fork_id,
            user_id,
            cartridge_id
        )
        .execute(&self.db)
        .await?;
        
        // Update cartridge fork count
        self.update_fork_count(cartridge_id).await?;
        
        Ok(true)
    }
    
    async fn update_star_count(&self, cartridge_id: &str) -> Result<()> {
        sqlx::query!(
            "UPDATE cartridges SET stars = (
                SELECT COUNT(*) FROM cartridge_stars WHERE cartridge_id = $1
             ) WHERE id = $1",
            cartridge_id
        )
        .execute(&self.db)
        .await?;
        
        Ok(())
    }
}

pub struct AnalyticsService {
    db: PgPool,
    clickhouse: ClickHouse,

impl AnalyticsService {
    pub async fn record_download(&self, cartridge_id: &str, user_id: Option<&str>) -> Result<()> {
        // Record in ClickHouse for analytics
        self.clickhouse.execute(
            "INSERT INTO cartridge_downloads (cartridge_id, user_id, timestamp)
             VALUES (?, ?, now())",
            &[cartridge_id.into(), user_id.unwrap_or("").into()]
        ).await?;
        
        // Update PostgreSQL counter
        sqlx::query!(
            "UPDATE cartridges SET 
                downloads = downloads + 1,
                weekly_downloads = weekly_downloads + 1,
                monthly_downloads = monthly_downloads + 1
             WHERE id = $1",
            cartridge_id
        )
        .execute(&self.db)
        .await?;
        
        Ok(())
    }
    
    pub async fn get_trending(&self, limit: usize) -> Result<Vec<TrendingCartridge>> {
        // Get cartridges with highest download growth
        let results = self.clickhouse.query(
            "SELECT 
                cartridge_id,
                COUNT(*) as downloads,
                COUNT(DISTINCT user_id) as unique_users
             FROM cartridge_downloads
             WHERE timestamp > now() - INTERVAL 7 DAY
             GROUP BY cartridge_id
             ORDER BY downloads DESC
             LIMIT ?",
            &[limit.into()]
        ).await?;
        
        Ok(results.into_iter().map(|row| TrendingCartridge {
            cartridge_id: row.cartridge_id,
            downloads: row.downloads,
            unique_users: row.unique_users,
            growth_rate: self.calculate_growth_rate(&row.cartridge_id).await?,
        }).collect())
    }
}

// src/services/verification.rs
pub struct VerificationService {
    db: PgPool,
    security_scanner: SecurityScanner,
}

impl VerificationService {
    pub async fn verify_cartridge(&self, cartridge_id: &str) -> Result<VerificationResult> {
        // Get cartridge details
        let cartridge = self.get_cartridge(cartridge_id).await?;
        
        // Level 1: Author verification
        let author_verified = self.check_author_verified(&cartridge.author_id).await?;
        
        // Level 2: Security scan
        let security_result = self.security_scanner.scan(&cartridge).await?;
        
        // Level 3: Community metrics
        let community_metrics = self.get_community_metrics(cartridge_id).await?;
        
        // Determine verification level
        let level = if !author_verified {
            0
        } else if !security_result.passed {
            1
        } else if community_metrics.stars < 10 || community_metrics.rating < 4.5 {
            2
        } else {
            3
        };
        
        // Update verification status
        self.update_verification_status(cartridge_id, level).await?;
        
        Ok(VerificationResult {
            level,
            author_verified,
            security_result,
            community_metrics,
        })
    }
    
    pub async fn request_professional_audit(&self, cartridge_id: &str) -> Result<AuditRequest> {
        // Create audit request
        let request_id = Uuid::new_v4().to_string();
        
        sqlx::query!(
            "INSERT INTO audit_requests (id, cartridge_id, status, requested_at)
             VALUES ($1, $2, 'pending', NOW())",
            request_id,
            cartridge_id
        )
        .execute(&self.db)
        .await?;
        
        Ok(AuditRequest {
            id: request_id,
            cartridge_id: cartridge_id.to_string(),
            status: "pending".to_string(),
            requested_at: Utc::now(),
        })
    }
}

// src/services/recommendations.rs
pub struct RecommendationService {
    db: PgPool,
    clickhouse: ClickHouse,
}

impl RecommendationService {
    pub async fn get_recommendations(&self, user_id: &str, limit: usize) -> Result<Vec<Cartridge>> {
        // Get user's deployed cartridges
        let user_cartridges = self.get_user_cartridges(user_id).await?;
        
        if user_cartridges.is_empty() {
            // Fallback to popular cartridges for new users
            return self.get_popular_cartridges(limit).await;
        }
        
        // Find users with similar cartridge usage
        let similar_users = self.find_similar_users(&user_cartridges).await?;
        
        // Get cartridges deployed by similar users but not by current user
        let recommendations = self.get_unused_cartridges(&user_cartridges, &similar_users).await?;
        
        // Rank by popularity among similar users
        let ranked = self.rank_by_similarity(&recommendations, &similar_users).await?;
        
        Ok(ranked.into_iter().take(limit).collect())
    }
    
    async fn find_similar_users(&self, user_cartridges: &[String]) -> Result<Vec<String>> {
        // Use Jaccard similarity on cartridge sets
        let result = self.clickhouse.query(
            "WITH user_cartridge_counts AS (
                SELECT 
                    user_id,
                    COUNT(*) as common_count
                FROM cartridge_deployments
                WHERE cartridge_id IN ?
                GROUP BY user_id
                HAVING COUNT(*) > 0
             )
             SELECT 
                user_id,
                common_count / (
                    SELECT COUNT(DISTINCT cartridge_id)
                    FROM cartridge_deployments
                    WHERE user_id = user_cartridge_counts.user_id
                ) as similarity
             FROM user_cartridge_counts
             ORDER BY similarity DESC
             LIMIT 100",
            &[user_cartridges.into()]
        ).await?;
        
        Ok(result.into_iter().map(|row| row.user_id).collect())
    }
}

// src/services/curation.rs
pub struct CurationService {
    db: PgPool,
}

impl CurationService {
    pub async fn get_featured_cartridges(&self, limit: usize) -> Result<Vec<FeaturedCartridge>> {
        // Get manually curated featured cartridges
        let curated = sqlx::query_as!(
            FeaturedCartridge,
            "SELECT c.*, fc.featured_at, fc.reason, fc.category
             FROM cartridges c
             JOIN featured_cartridges fc ON c.id = fc.cartridge_id
             WHERE fc.is_active = true
             ORDER BY fc.featured_at DESC
             LIMIT $1",
            limit as i64
        )
        .fetch_all(&self.db)
        .await?;
        
        // Fallback: if not enough curated, add top rated
        if curated.len() < limit {
            let needed = limit - curated.len();
            let top_rated = self.get_top_rated(needed).await?;
            curated.extend(top_rated);
        }
        
        Ok(curated)
    }
    
    pub async fn feature_cartridge(
        &self,
        cartridge_id: &str,
        reason: &str,
        category: &str
    ) -> Result<()> {
        sqlx::query!(
            "INSERT INTO featured_cartridges 
                (cartridge_id, featured_at, reason, category, is_active)
             VALUES ($1, NOW(), $2, $3, true)
             ON CONFLICT (cartridge_id) DO UPDATE SET
                reason = EXCLUDED.reason,
                category = EXCLUDED.category,
                is_active = true,
                featured_at = NOW()",
            cartridge_id,
            reason,
            category
        )
        .execute(&self.db)
        .await?;
        
        Ok(())
    }
}

// frontend/components/Marketplace.jsx
import React, { useState, useEffect } from 'react';
import { searchCartridges, getPopular, getTrending, getVerified } from '../api/marketplace';

const Marketplace = () => {
  const [activeTab, setActiveTab] = useState('popular');
  const [cartridges, setCartridges] = useState([]);
  const [loading, setLoading] = useState(false);
  const [searchQuery, setSearchQuery] = useState('');

  useEffect(() => {
    fetchCartridges();
  }, [activeTab]);

  const fetchCartridges = async () => {
    setLoading(true);
    try {
      let results;
      switch (activeTab) {
        case 'popular':
          results = await getPopular();
          break;
        case 'trending':
          results = await getTrending();
          break;
        case 'verified':
          results = await getVerified();
          break;
        default:
          results = await searchCartridges(searchQuery);
      }
      setCartridges(results);
    } catch (error) {
      console.error('Failed to fetch cartridges:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleSearch = (e) => {
    e.preventDefault();
    setActiveTab('search');
    fetchCartridges();
  };

  return (
    <div className="marketplace">
      <h1>🛒 Cartridge Marketplace</h1>
      
      <div className="search-bar">
        <form onSubmit={handleSearch}>
          <input
            type="text"
            placeholder="Search cartridges..."
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
          />
          <button type="submit">🔍 Search</button>
        </form>
      </div>

      <div className="tabs">
        <button
          className={activeTab === 'popular' ? 'active' : ''}
          onClick={() => setActiveTab('popular')}
        >
          🌟 Popular
        </button>
        <button
          className={activeTab === 'trending' ? 'active' : ''}
          onClick={() => setActiveTab('trending')}
        >
          📈 Trending
        </button>
        <button
          className={activeTab === 'verified' ? 'active' : ''}
          onClick={() => setActiveTab('verified')}
        >
          ✅ Verified
        </button>
      </div>

      {loading ? (
        <div className="loading">Loading cartridges...</div>
      ) : (
        <div className="cartridge-grid">
          {cartridges.map(cartridge => (
            <CartridgeCard key={cartridge.id} cartridge={cartridge} />

        </div>
      )}
    </div>
  );
};

// src/routes/deployments.rs
#[actix_web::post("/deployments")]
async fn create_deployment(
    req: HttpRequest,
    deployment_data: web::Json<DeploymentRequest>,
    user: ReqData<User>
) -> Result<HttpResponse> {
    // Validate user can deploy this cartridge
    let cartridge = cartridge_service.get_cartridge(&deployment_data.cartridge_id)
        .await?;
    
    if !cartridge.is_public && !user.can_access(&cartridge) {
        return Err(DeploymentError::Unauthorized);
    }
    
    // Create deployment record
    let deployment_id = deployment_service.create_deployment(
        &user.id,
        &deployment_data.cartridge_id,
        deployment_data.config.clone()
    ).await?;
    
    // Trigger provisioning
    provisioner.deploy_cartridge(
        &deployment_id,
        &cartridge,
        deployment_data.config
    ).await?;
    
    // Record analytics
    analytics_service.record_deployment(
        &user.id,
        &cartridge.id,
        &deployment_id
    ).await?;
    
    Ok(HttpResponse::Created().json(DeploymentResponse {
        id: deployment_id,
        status: "provisioning".to_string(),
        cartridge: cartridge.into(),
        created_at: Utc::now(),
    }))
}

}

**Recommendation Algorithm:**

# Pseudocode for recommendation algorithm
def recommend_cartridges(user_id, limit=10):
    # Get user's deployed cartridges
    user_cartridges = get_user_cartridges(user_id)
    
    if not user_cartridges:
        # New user - return popular cartridges
        return get_popular_cartridges(limit)
    
    # Find similar users (Jaccard similarity)
    similar_users = find_similar_users(user_cartridges)
    
    # Get cartridges used by similar users but not by current user
    candidate_cartridges = get_unused_cartridges(user_cartridges, similar_users)
    
    # Rank by:
    # 1. Frequency among similar users
    # 2. Overall popularity
    # 3. Rating
    # 4. Recency
    ranked = rank_cartridges(candidate_cartridges, similar_users)
    
    # Add some diversity
    diverse_ranked = add_diversity(ranked)
    
    return diverse_ranked[:limit]

def find_similar_users(user_cartridges, top_n=100):
    # Calculate Jaccard similarity between users
    # J(A,B) = |A ∩ B| / |A ∪ B|
    similarities = []
    
    for user in all_users:
        if user == user_id:
            continue
            
        user_set = set(get_user_cartridges(user))
        current_set = set(user_cartridges)
        
        intersection = len(user_set & current_set)
        union = len(user_set | current_set)
        
        if union > 0:
            similarity = intersection / union
            similarities.append((user, similarity))
    
    # Return top N most similar users
    return sorted(similarities, key=lambda x: x[1], reverse=True)[:top_n]
