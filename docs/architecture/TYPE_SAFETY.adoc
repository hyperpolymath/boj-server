### 3. Runtime Type Checking

[source,rust]
----
// Runtime type validation for FFI
fn validate_ffi_type<T>(value: &ZigValue) -> Result<T, FFIError> 
where
    T: FromZig + TypeChecked,
{
    // 1. Check type tag
    if !value.has_type_tag::<T>() {
        return Err(FFIError::TypeMismatch);
    }
    
    // 2. Validate memory safety
    if !value.is_memory_safe() {
        return Err(FFIError::MemoryViolation);
    }
    
    // 3. Convert safely
    T::from_zig(value)
}

// Example usage
let result = validate_ffi_type::<String>(zig_value)?;
----
